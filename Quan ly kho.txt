function getInitAppData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  // --- BẢNG DATA_THIET_BI ---
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const tbData = (tbSheet && tbSheet.getLastRow() > 1) ? tbSheet.getRange(2, 1, tbSheet.getLastRow() - 1, 17).getValues() : [];
  
  const tonKhoList = [], baoHanhList = [], existingSerials = [];
  const tbMap = {};

  tbData.forEach((r, idx) => {
    const sn = String(r[0] || '').trim().toUpperCase();
    if (sn) existingSerials.push(sn);

    let ngayNhapDate = null, ngayNhapStr = "--/--/----";
    if (r[7] instanceof Date && !isNaN(r[7].getTime())) {
      ngayNhapDate = r[7]; ngayNhapStr = Utilities.formatDate(r[7], "GMT+7", "dd/MM/yyyy");
    } else if (r[7]) {
      ngayNhapStr = String(r[7]).trim();
      if (ngayNhapStr.includes('-')) { const p = ngayNhapStr.split('-'); ngayNhapDate = new Date(p[0], p[1] - 1, p[2]); } 
      else if (ngayNhapStr.includes('/')) { const p = ngayNhapStr.split('/'); ngayNhapDate = new Date(p[2], p[1] - 1, p[0]); }
    }

    if (sn) { tbMap[sn] = { model: r[1] || '-', tenHang: r[2] || '-', hanBh: r[15] instanceof Date ? Utilities.formatDate(r[15], "GMT+7", "dd/MM/yyyy") : (r[15] || '-') }; }

    if (String(r[9]).trim() === "Tồn kho") {
      let soNgayLuuKho = 0;
      if (ngayNhapDate && !isNaN(ngayNhapDate.getTime())) {
        ngayNhapDate.setHours(0, 0, 0, 0);
        soNgayLuuKho = Math.floor((today - ngayNhapDate) / (1000 * 60 * 60 * 24));
      }
      tonKhoList.push({ rowId: idx + 2, serial: sn, model: String(r[1] || ''), tenHang: String(r[2] || ''), nhomHang: String(r[3] || ''), loaiHang: String(r[4] || ''), kho: String(r[5] || ''), ncc: String(r[6] || ''), ngayNhap: ngayNhapStr, maPhieu: String(r[8] || ''), ghiChu: String(r[16] || ''), soNgayLuuKho: Math.max(0, soNgayLuuKho) });
    }

    if (String(r[9]).trim() === "Đã xuất" || r[15]) {
      let expDate = null, expStr = "--/--/----";
      if (r[15] instanceof Date && !isNaN(r[15].getTime())) { expDate = r[15]; expStr = Utilities.formatDate(r[15], "GMT+7", "dd/MM/yyyy"); } 
      else if (r[15]) { expStr = String(r[15]).trim(); if (expStr.includes('/')) { const p = expStr.split('/'); expDate = new Date(p[2], p[1] - 1, p[0]); } }

      let diffDays = -999, statusColor = "RED", statusText = "Hết hạn";
      if (expDate && !isNaN(expDate.getTime())) {
        expDate.setHours(0, 0, 0, 0);
        diffDays = Math.ceil((expDate - today) / (1000 * 60 * 60 * 24));
        if (diffDays > 30) { statusColor = "GREEN"; statusText = `Còn ${diffDays} ngày`; } 
        else if (diffDays >= 0) { statusColor = "YELLOW"; statusText = `Còn ${diffDays} ngày (Gấp)`; } 
        else { statusColor = "RED"; statusText = `Quá hạn ${Math.abs(diffDays)} ngày`; }
      }

      let ngayXuatStr = "--/--/----";
      if (r[10] instanceof Date) ngayXuatStr = Utilities.formatDate(r[10], "GMT+7", "dd/MM/yyyy");
      else if (r[10]) ngayXuatStr = String(r[10]);

      baoHanhList.push({ serial: sn, model: String(r[1] || ''), tenHang: String(r[2] || ''), nhomHang: String(r[3] || ''), loaiHang: String(r[4] || ''), kho: String(r[5] || ''), ncc: String(r[6] || ''), ngayNhap: ngayNhapStr, ngayXuat: ngayXuatStr, maPhieuXuat: String(r[11] || ''), khachHang: String(r[12] || ''), sdtKhach: String(r[13] || ''), soThangBh: r[14] || '', ngayHetHanBh: expStr, statusColor, statusText });
    }
  });

  // --- CÁC BẢNG DANH MỤC ---
  const spSheet = ss.getSheetByName("DM_SAN_PHAM");
  const products = (spSheet && spSheet.getLastRow() > 1) ? spSheet.getRange(2, 1, spSheet.getLastRow() - 1, 3).getValues().map((r, i) => ({ rowId: i + 2, model: r[0], ten: r[1], nhom: r[2] })) : [];

  const nccSheet = ss.getSheetByName("DM_NCC");
  const ncc = (nccSheet && nccSheet.getLastRow() > 1) ? nccSheet.getRange(2, 1, nccSheet.getLastRow() - 1, 4).getValues().map((r, i) => ({ rowId: i + 2, tenTat: r[0], tenDayDu: r[1], sdt: r[2], ghiChu: r[3] })) : [];

  const khSheet = ss.getSheetByName("DM_KHACH_HANG");
  const khachHang = (khSheet && khSheet.getLastRow() > 1) ? khSheet.getRange(2, 1, khSheet.getLastRow() - 1, 4).getValues().map((r, i) => ({ rowId: i + 2, ten: r[0], sdt: r[1], diaChi: r[2], ghiChu: r[3] })) : [];

  const qcSheet = ss.getSheetByName("DM_QUY_CHUAN");
  const nhomHang = [], kho = [], loaiHang = [], baoHanh = [];
  if (qcSheet && qcSheet.getLastRow() > 1) {
    qcSheet.getRange(2, 1, qcSheet.getLastRow() - 1, 4).getValues().forEach((r, i) => {
      const rowId = i + 2;
      if (r[0]) nhomHang.push({ rowId, col: 1, val: r[0] });
      if (r[1]) kho.push({ rowId, col: 2, val: r[1] });
      if (r[2]) loaiHang.push({ rowId, col: 3, val: r[2] });
      if (r[3]) baoHanh.push({ rowId, col: 4, val: r[3] });
    });
  }

  // --- LỊCH SỬ NHẬP (TỰ ĐỘNG TÌM TÊN HÀNG ĐẦY ĐỦ THÔNG MINH) ---
  const lsNhapSheet = ss.getSheetByName("LICH_SU_NHAP");
  const lsNhapData = (lsNhapSheet && lsNhapSheet.getLastRow() > 1) ? lsNhapSheet.getRange(2, 1, lsNhapSheet.getLastRow() - 1, 8).getValues() : [];
  
  const lsNhap = lsNhapData.map((r, i) => {
    const rawModel = String(r[3] || '');
    // Thuật toán gọt bỏ phần số lượng trong ngoặc (Ví dụ: "Canon 6030w (5)" -> "Canon 6030w")
    const cleanModelKey = rawModel.replace(/\s*\(\d+\)/g, '').trim().toUpperCase();
    
    // Tìm tên hàng hóa đầy đủ từ danh mục
    let matchedTenHang = '-';
    for (let p of products) {
      if (String(p.model).trim().toUpperCase() === cleanModelKey) {
        matchedTenHang = p.ten;
        break;
      }
    }

    // Nếu là lô đa thiết bị gồm nhiều model (có dấu +)
    if (rawModel.includes('+')) {
      matchedTenHang = 'Lô hàng tổng hợp nhiều mặt hàng';
    }

    return {
      id: i + 1,
      maPhieu: r[0],
      ngayNhap: r[1] instanceof Date ? Utilities.formatDate(r[1], "GMT+7", "dd/MM/yyyy") : r[1],
      ncc: r[2] || 'Chưa gán',
      model: rawModel,
      tenHang: matchedTenHang,
      soLuong: r[4] || 1,
      serials: r[5],
      kho: r[6] || 'Kho VP',
      ghiChu: r[7] || ''
    };
  }).reverse();

  // --- LỊCH SỬ XUẤT ---
  const lsXuatSheet = ss.getSheetByName("LICH_SU_XUAT");
  const lsXuatData = (lsXuatSheet && lsXuatSheet.getLastRow() > 1) ? lsXuatSheet.getRange(2, 1, lsXuatSheet.getLastRow() - 1, 7).getValues() : [];
  const lsXuat = lsXuatData.map((r, i) => {
    const rawSerials = String(r[4] || '');
    const firstSn = rawSerials.split(',')[0].trim().toUpperCase();
    const spInfo = tbMap[firstSn] || { model: 'Thiết bị', tenHang: 'Theo đơn hàng', hanBh: '-' };
    return { id: i + 1, maPhieu: r[0], ngayXuat: r[1] instanceof Date ? Utilities.formatDate(r[1], "GMT+7", "dd/MM/yyyy") : r[1], khachHang: r[2], soLuong: r[3], serials: rawSerials, model: spInfo.model, tenHang: spInfo.tenHang, baoHanh: r[5], hanBhDen: spInfo.hanBh, ghiChu: r[6] };
  }).reverse();

  return { tonKhoList, baoHanhList, existingSerials, products, ncc, khachHang, nhomHang, kho, loaiHang, baoHanh, lsNhap, lsXuat };
}

function generateAutoSerials(count) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const cfgSheet = ss.getSheetByName("CAU_HINH");
  
  let prefix = "TA-";
  if (cfgSheet && cfgSheet.getLastRow() > 1) {
    const cData = cfgSheet.getRange(2, 1, cfgSheet.getLastRow() - 1, 2).getValues();
    const f = cData.find(r => r[0] === 'TIEN_TO_SERIAL');
    if (f) prefix = f[1];
  }

  const today = new Date();
  const yy = String(today.getFullYear()).slice(-2);
  const mm = String(today.getMonth() + 1).padStart(2, '0');
  const dd = String(today.getDate()).padStart(2, '0');
  const datePrefix = `${prefix}${yy}${mm}${dd}-`;

  let maxIndex = 0;
  if (tbSheet && tbSheet.getLastRow() > 1) {
    const serials = tbSheet.getRange(2, 1, tbSheet.getLastRow() - 1, 1).getValues();
    serials.forEach(r => {
      const sn = String(r[0]).trim();
      if (sn.startsWith(datePrefix)) {
        const numPart = parseInt(sn.replace(datePrefix, ''), 10);
        if (!isNaN(numPart) && numPart > maxIndex) maxIndex = numPart;
      }
    });
  }

  const generated = [];
  for (let i = 1; i <= Number(count); i++) {
    generated.push(`${datePrefix}${String(maxIndex + i).padStart(3, '0')}`);
  }
  return generated;
}

function executeNhapKhoMulti(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const lsSheet = ss.getSheetByName("LICH_SU_NHAP");

  if (!data.items || data.items.length === 0) throw new Error("Chưa có mặt hàng nào!");

  const allCleanSerials = [];
  const internalCheck = new Set();
  data.items.forEach(item => {
    item.serials.forEach(s => {
      const sn = String(s).trim().toUpperCase();
      if (!sn) return;
      if (internalCheck.has(sn)) throw new Error(`Serial [${sn}] bị trùng lặp ngay trong phiếu!`);
      internalCheck.add(sn);
      allCleanSerials.push({ sn, item });
    });
  });

  const lastRow = tbSheet.getLastRow();
  if (lastRow > 1) {
    const existing = tbSheet.getRange(2, 1, lastRow - 1, 1).getValues().map(r => String(r[0]).trim().toUpperCase());
    const dupes = allCleanSerials.filter(obj => existing.includes(obj.sn));
    if (dupes.length > 0) throw new Error(`Các Serial sau đã tồn tại: ${dupes.map(d => d.sn).join(', ')}`);
  }

  const newRows = allCleanSerials.map(obj => [ obj.sn, obj.item.model, obj.item.tenHang, obj.item.nhomHang, data.loaiHang, data.kho, data.ncc, data.ngayNhap, data.maPhieu, "Tồn kho", "", "", "", "", "", "", data.ghiChu ]);
  tbSheet.getRange(lastRow + 1, 1, newRows.length, 17).setValues(newRows);

  const allSnStr = allCleanSerials.map(d => d.sn).join(', ');
  const modelSummary = data.items.map(i => `${i.model} (${i.serials.length})`).join(' + ');
  lsSheet.appendRow([ data.maPhieu, data.ngayNhap, data.ncc, modelSummary, allCleanSerials.length, allSnStr, data.kho, data.ghiChu ]);

  return `Đã nhập thành công lô hàng ${allCleanSerials.length} thiết bị!`;
}

function updateThietBi(rowId, serial, model, kho, ncc, ghiChu) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const spSheet = ss.getSheetByName("DM_SAN_PHAM");

  const cleanSerial = String(serial).trim().toUpperCase();
  const targetRow = Number(rowId);

  const allSerials = tbSheet.getRange(2, 1, tbSheet.getLastRow() - 1, 1).getValues();
  for (let i = 0; i < allSerials.length; i++) {
    if (String(allSerials[i][0]).trim().toUpperCase() === cleanSerial && (i + 2) !== targetRow) {
      throw new Error(`Mã Serial [${cleanSerial}] đã tồn tại trên một thiết bị khác!`);
    }
  }

  let tenHang = "", nhomHang = "";
  if (spSheet && spSheet.getLastRow() > 1) {
    const spData = spSheet.getRange(2, 1, spSheet.getLastRow() - 1, 3).getValues();
    for (let r of spData) {
      if (String(r[0]).toUpperCase() === String(model).toUpperCase()) { tenHang = r[1]; nhomHang = r[2]; break; }
    }
  }

  tbSheet.getRange(targetRow, 1).setValue(cleanSerial);
  tbSheet.getRange(targetRow, 2, 1, 3).setValues([[model, tenHang, nhomHang]]);
  tbSheet.getRange(targetRow, 6, 1, 2).setValues([[kho, ncc]]);
  tbSheet.getRange(targetRow, 17).setValue(ghiChu);

  writeAuditLog("SỬA MÁY TRONG KHO", cleanSerial, `Đổi thông tin Model: ${model}, Kho: ${kho}`);
  return "Cập nhật thiết bị thành công!";
}

  return "Đã xóa thiết bị khỏi kho thành công!";
}

function executeXuatKho(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const lsSheet = ss.getSheetByName("LICH_SU_XUAT");

  if (!data.serials || data.serials.length === 0) throw new Error("Chưa chọn máy nào!");

  const cleanSerials = data.serials.map(s => String(s).trim().toUpperCase());
  const allTbData = tbSheet.getRange(2, 1, tbSheet.getLastRow() - 1, 10).getValues();
  const rowsToUpdate = [];

  cleanSerials.forEach(sn => {
    let found = false;
    for (let i = 0; i < allTbData.length; i++) {
      if (String(allTbData[i][0]).trim().toUpperCase() === sn) {
        if (String(allTbData[i][9]).trim() !== "Tồn kho") throw new Error(`Serial [${sn}] không còn tồn kho!`);
        rowsToUpdate.push(i + 2);
        found = true; break;
      }
    }
    if (!found) throw new Error(`Không tìm thấy mã Serial [${sn}]!`);
  });

  const ngayXuatParts = data.ngayXuat.split('-');
  const expDate = new Date(ngayXuatParts[0], ngayXuatParts[1] - 1, ngayXuatParts[2]);
  const soThang = parseInt(data.soThangBh.replace(/\D/g, ''), 10) || 12;
  expDate.setMonth(expDate.getMonth() + soThang);

  const ngayXuatFormat = Utilities.formatDate(new Date(data.ngayXuat), "GMT+7", "dd/MM/yyyy");
  const ngayHetHanFormat = Utilities.formatDate(expDate, "GMT+7", "dd/MM/yyyy");

  rowsToUpdate.forEach(rowId => {
    tbSheet.getRange(rowId, 10).setValue("Đã xuất");
    tbSheet.getRange(rowId, 11, 1, 6).setValues([[ ngayXuatFormat, data.maPhieu, data.tenKhach, data.sdtKhach, soThang, ngayHetHanFormat ]]);
    if (data.ghiChu) tbSheet.getRange(rowId, 17).setValue(data.ghiChu);
  });

  lsSheet.appendRow([ data.maPhieu, ngayXuatFormat, `${data.tenKhach} (${data.sdtKhach})`, cleanSerials.length, cleanSerials.join(', '), data.soThangBh, data.ghiChu ]);
  return `Xuất kho thành công ${cleanSerials.length} thiết bị! Đã kích hoạt bảo hành.`;
}

function saveProduct(model, ten, nhom, rowId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("DM_SAN_PHAM");
  model = String(model).trim(); ten = String(ten).trim(); nhom = String(nhom).trim();
  if (!model || !ten || !nhom) throw new Error("Điền đủ Model, Tên, Nhóm!");

  if (rowId) {
    sheet.getRange(Number(rowId), 1, 1, 3).setValues([[model, ten, nhom]]);
    return "Cập nhật sản phẩm thành công!";
  } else {
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) { if (String(data[i][0]).toUpperCase() === model.toUpperCase()) throw new Error("Mã Model đã tồn tại!"); }
    sheet.appendRow([model, ten, nhom]);
    return "Thêm Model mới thành công!";
  }
}
function deleteProduct(rowId) { SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DM_SAN_PHAM").deleteRow(Number(rowId)); return "Đã xóa Model!"; }

function addQuyChuan(colIndex, val) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DM_QUY_CHUAN");
  if (!val) throw new Error("Không được để trống!");
  const colValues = sheet.getRange(1, colIndex, sheet.getMaxRows(), 1).getValues();
  let targetRow = 1; while (targetRow <= colValues.length && colValues[targetRow - 1][0] !== "") targetRow++;
  sheet.getRange(targetRow, colIndex).setValue(val);
  return "Thêm thành công!";
}
function deleteQuyChuan(rowId, colIndex) { SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DM_QUY_CHUAN").getRange(Number(rowId), Number(colIndex)).clearContent(); return "Đã xóa!"; }

function getCaiDatData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  // Tab CAU_HINH
  const cfgSheet = ss.getSheetByName("CAU_HINH");
  const configs = {};
  if (cfgSheet && cfgSheet.getLastRow() > 1) {
    cfgSheet.getRange(2, 1, cfgSheet.getLastRow() - 1, 2).getValues().forEach(r => { configs[r[0]] = r[1]; });
  }

  // TỰ ĐỘNG TẠO TÀI KHOẢN ADMIN NẾU TRỐNG (AUTO-SEEDING)
  let userSheet = ss.getSheetByName("USERS");
  if (!userSheet) {
    userSheet = ss.insertSheet("USERS");
    userSheet.appendRow(["Tên đăng nhập", "Mật khẩu", "Họ và tên", "Vai trò", "Trạng thái"]);
    userSheet.getRange(1, 1, 1, 5).setFontWeight("bold").setBackground("#e2e8f0");
  }
  if (userSheet.getLastRow() <= 1) {

  function renderCharts(currentMonth, currentYear, rangeMonths = 6) {
    // 1. TÍNH TOÁN CÁC MỐC THÁNG THEO DROPDOWN CHỌN
    const labels = [];
    const nhapData = new Array(rangeMonths).fill(0);
    const xuatData = new Array(rangeMonths).fill(0);

    for (let i = rangeMonths - 1; i >= 0; i--) {
      let m = currentMonth - i;
      let y = currentYear;
      if (m <= 0) { 
        m += 12; 
        y -= 1; 
      }
      labels.push(`T${m}/${y}`);
    }

    // Gom số lượng Nhập theo mốc tháng đã chọn
    lsNhapList.forEach(pn => {
      if (pn.ngayNhap && pn.ngayNhap.includes('/')) {
        const parts = pn.ngayNhap.split('/');
        const tag = `T${parseInt(parts[1], 10)}/${parseInt(parts[2], 10)}`;
        const idx = labels.indexOf(tag);
        if (idx !== -1) nhapData[idx] += Number(pn.soLuong || 1);
      }
    });

    // Gom số lượng Xuất theo mốc tháng đã chọn
    lsXuatList.forEach(px => {
      if (px.ngayXuat && px.ngayXuat.includes('/')) {
        const parts = px.ngayXuat.split('/');
        const tag = `T${parseInt(parts[1], 10)}/${parseInt(parts[2], 10)}`;
        const idx = labels.indexOf(tag);
        if (idx !== -1) xuatData[idx] += Number(px.soLuong || 1);
      }
    });

    // Vẽ lại Biểu đồ Cột
    const ctxBar = document.getElementById('chartNhapXuat');
    if (ctxBar) {
      if (chartNhapXuatInstance) chartNhapXuatInstance.destroy();
      chartNhapXuatInstance = new Chart(ctxBar, {
        type: 'bar',
        data: {
          labels: labels,
          datasets: [
            { label: 'Số lượng Nhập', data: nhapData, backgroundColor: '#0d6efd', borderRadius: 4 },
            { label: 'Số lượng Xuất', data: xuatData, backgroundColor: '#dc3545', borderRadius: 4 }
          ]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: { legend: { position: 'top' } }
        }
      });
    }

    // 2. VẼ BIỂU ĐỒ TRÒN (NHÓM HÀNG)
    const nhomCounts = {};
    tonKhoList.forEach(item => {
      const g = item.nhomHang || 'Khác';
      nhomCounts[g] = (nhomCounts[g] || 0) + 1;
    });

    const donutLabels = Object.keys(nhomCounts);
    const donutData = Object.values(nhomCounts);
    const donutColors = ['#0d6efd', '#198754', '#ffc107', '#0dcaf0', '#6c757d', '#d63384'];

    const ctxPie = document.getElementById('chartNhomHang');
    if (ctxPie) {
      if (chartNhomHangInstance) chartNhomHangInstance.destroy();
      chartNhomHangInstance = new Chart(ctxPie, {
        type: 'doughnut',
        data: {
          labels: donutLabels.length > 0 ? donutLabels : ['Trống'],
          datasets: [{
            data: donutData.length > 0 ? donutData : [1],
            backgroundColor: donutLabels.length > 0 ? donutColors.slice(0, donutLabels.length) : ['#e9ecef']
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: { legend: { position: 'bottom' } }
        }
      });
    }
  }

  // RENDER BẢNG LỊCH SỬ NHẬP (GỌT SẠCH 100% SỐ TRONG NGOẶC)
  function renderLsNhapTable(data) {
    document.getElementById('ls-nhap-total').innerText = `${data.length} đợt nhập`;
    const tbody = document.getElementById('ls-nhap-table-body');
    if (!tbody) return;

    if (data.length === 0) {
      tbody.innerHTML = '<tr><td colspan="8" class="text-center text-muted py-3">Không tìm thấy lịch sử nhập kho.</td></tr>';
      return;
    }

    tbody.innerHTML = data.map(item => {
      const rawModel = String(item.model || '');
      let modelContent = '';
      let tenHangContent = '';

      // 1. Nếu là LÔ NHIỀU MẶT HÀNG (Có dấu +)
      if (rawModel.includes('+')) {
        const parts = rawModel.split('+');
        modelContent = parts.map(p => {
          const match = p.trim().match(/^(.+?)\s*\(\d+\)$/);
          const cleanM = match ? match[1].trim() : p.trim();
          return `<div class="mb-1"><strong class="text-primary">${cleanM}</strong></div>`;
        }).join('');
        tenHangContent = item.tenHang;
      } 
      // 2. Nếu là LÔ 1 MODEL (Gọt sạch số trong ngoặc như (1), (5)...)
      else {
        const cleanModel = rawModel.replace(/\s*\(\d+\)/g, '').trim();
        modelContent = `<strong class="text-primary">${cleanModel}</strong>`;
        tenHangContent = item.tenHang;
      }

      return `
        <tr>
          <td class="fw-bold font-monospace text-primary">${item.maPhieu}</td>
          <td>${item.ngayNhap}</td>
          <td class="fw-bold">${item.ncc}</td>
          <td>${modelContent}</td>
          <td>${tenHangContent}</td>
          <td class="text-center fw-bold text-success fs-6">${item.soLuong}</td>
          <td><span class="badge bg-info text-dark">${item.kho}</span></td>
          <td class="text-center">
            <button class="btn btn-sm btn-outline-primary py-1 px-2 fw-bold" onclick="openNhapBreakdown('${item.maPhieu}', '${item.ncc}', '${item.ngayNhap}')" title="Bóc tách chi tiết từng cái máy">
              <i class="fa-solid fa-boxes-packing"></i> Bóc tách
            </button>
          </td>
        </tr>
      `;
    }).join('');
  }

// 1. Đọc dữ liệu từ các sheet riêng biệt
function getMasterData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const getSheetData = (name, colCount) => {
    const sheet = ss.getSheetByName(name);
    if (!sheet || sheet.getLastRow() <= 1) return [];
    return sheet.getRange(2, 1, sheet.getLastRow() - 1, colCount).getValues();
  };

  // 1. Lấy Sản phẩm
  const spRows = getSheetData("DM_SAN_PHAM", 3);
  const products = spRows.map((r, i) => ({ rowId: i + 2, model: r[0], ten: r[1], nhom: r[2] }));

  // 2. Lấy Nhà cung cấp (Tên tắt, Tên đầy đủ, SĐT, Ghi chú)
  const nccRows = getSheetData("DM_NCC", 4);
  const ncc = nccRows.map((r, i) => ({ rowId: i + 2, tenTat: r[0], tenDayDu: r[1], sdt: r[2], ghiChu: r[3] }));

  // 3. Lấy Khách hàng
  const khRows = getSheetData("DM_KHACH_HANG", 4);
  const khachHang = khRows.map((r, i) => ({ rowId: i + 2, ten: r[0], sdt: r[1], diaChi: r[2], ghiChu: r[3] }));

  // 4. Lấy Quy chuẩn (4 cột độc lập)
  const qcRows = getSheetData("DM_QUY_CHUAN", 4);
  const nhomHang = [], kho = [], loaiHang = [], baoHanh = [];
  qcRows.forEach((r, i) => {
    const rowId = i + 2;
    if (r[0]) nhomHang.push({ rowId: rowId, col: 1, val: r[0] });
    if (r[1]) kho.push({ rowId: rowId, col: 2, val: r[1] });
    if (r[2]) loaiHang.push({ rowId: rowId, col: 3, val: r[2] });
    if (r[3]) baoHanh.push({ rowId: rowId, col: 4, val: r[3] });
  });

  return { products, ncc, khachHang, nhomHang, kho, loaiHang, baoHanh };
}

// 2. XỬ LÝ MODEL SẢN PHẨM (Thêm / Sửa / Xóa)
function saveProduct(model, ten, nhom, rowId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("DM_SAN_PHAM");
  model = String(model).trim();
  ten = String(ten).trim();
  nhom = String(nhom).trim();

  if (!model || !ten || !nhom) throw new Error("Vui lòng điền đủ: Model, Tên máy và Nhóm hàng!");

  if (rowId) {
    // Chế độ SỬA
    sheet.getRange(Number(rowId), 1, 1, 3).setValues([[model, ten, nhom]]);
    return "Cập nhật sản phẩm thành công!";
  } else {
    // Chế độ THÊM MỚI
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).toUpperCase() === model.toUpperCase()) {
        throw new Error("Mã Model này đã tồn tại trong danh mục!");
      }
    }
    sheet.appendRow([model, ten, nhom]);
    return "Thêm Model mới thành công!";
  }
}

function deleteProduct(rowId) {
  SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DM_SAN_PHAM").deleteRow(Number(rowId));
  return "Đã xóa Model!";
}

function addQuyChuan(colIndex, val) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("DM_QUY_CHUAN");
  val = String(val).trim();
  if (!val) throw new Error("Không được để trống giá trị!");

  const colValues = sheet.getRange(1, colIndex, sheet.getMaxRows(), 1).getValues();
  let targetRow = 1;
  while (targetRow <= colValues.length && colValues[targetRow - 1][0] !== "") {
    targetRow++;
  }
  sheet.getRange(targetRow, colIndex).setValue(val);
  return "Thêm thành công!";
}

function deleteQuyChuan(rowId, colIndex) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("DM_QUY_CHUAN");
  sheet.getRange(Number(rowId), Number(colIndex)).clearContent();
  return "Đã xóa!";
}

// 2. Hàm Tự sinh mã Serial theo chuẩn: TA-YYMMDD-XXX
function generateAutoSerials(count) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const lastRow = tbSheet.getLastRow();
  
  const today = new Date();
  const yy = String(today.getFullYear()).slice(-2);
  const mm = String(today.getMonth() + 1).padStart(2, '0');
  const dd = String(today.getDate()).padStart(2, '0');
  const datePrefix = `TA-${yy}${mm}${dd}-`;

  let maxIndex = 0;
  if (lastRow > 1) {
    const serials = tbSheet.getRange(2, 1, lastRow - 1, 1).getValues();
    serials.forEach(r => {
      const sn = String(r[0]).trim();
      if (sn.startsWith(datePrefix)) {
        const numPart = parseInt(sn.replace(datePrefix, ''), 10);
        if (!isNaN(numPart) && numPart > maxIndex) maxIndex = numPart;
      }
    });
  }

  const generated = [];
  for (let i = 1; i <= Number(count); i++) {
    const nextNum = String(maxIndex + i).padStart(3, '0');
    generated.push(`${datePrefix}${nextNum}`);
  }
  return generated;
}

// Thực thi Nhập kho Đa Model / Nhiều mặt hàng trong 1 phiếu
function executeNhapKhoMulti(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const lsSheet = ss.getSheetByName("LICH_SU_NHAP");

  if (!data.items || data.items.length === 0) {
    throw new Error("Chưa có mặt hàng nào trong phiếu nhập!");
  }

  // 1. Quét kiểm tra trùng Serial trên toàn bộ các món hàng
  const allCleanSerials = [];
  const internalCheck = new Set();

  data.items.forEach(item => {
    item.serials.forEach(s => {
      const sn = String(s).trim().toUpperCase();
      if (!sn) return;
      if (internalCheck.has(sn)) {
        throw new Error(`Serial [${sn}] bị trùng lặp ngay trong phiếu nhập này!`);
      }
      internalCheck.add(sn);
      allCleanSerials.push({ sn, item });
    });
  });

  if (allCleanSerials.length === 0) throw new Error("Chưa có mã Serial nào được nhập!");

  // Check trùng với CSDL Sheet
  const lastRow = tbSheet.getLastRow();
  if (lastRow > 1) {
    const existing = tbSheet.getRange(2, 1, lastRow - 1, 1).getValues().map(r => String(r[0]).trim().toUpperCase());
    const dupes = allCleanSerials.filter(obj => existing.includes(obj.sn));
    if (dupes.length > 0) {
      throw new Error(`Các mã Serial sau đã tồn tại trong kho: ${dupes.map(d => d.sn).join(', ')}`);
    }
  }

  // 2. Chèn từng máy vào DATA_THIET_BI
  const newRows = allCleanSerials.map(obj => [
    obj.sn, obj.item.model, obj.item.tenHang, obj.item.nhomHang, data.loaiHang, data.kho,
    data.ncc, data.ngayNhap, data.maPhieu, "Tồn kho", "", "", "", "", "", "", data.ghiChu
  ]);
  tbSheet.getRange(lastRow + 1, 1, newRows.length, 17).setValues(newRows);

  // 3. Ghi vào LICH_SU_NHAP
  const allSnStr = allCleanSerials.map(d => d.sn).join(', ');
  const modelSummary = data.items.map(i => `${i.model} (${i.serials.length})`).join(' + ');

  lsSheet.appendRow([
    data.maPhieu, data.ngayNhap, data.ncc, modelSummary,
    allCleanSerials.length, allSnStr, data.kho, data.ghiChu
  ]);

  return `Đã nhập thành công lô hàng ${allCleanSerials.length} thiết bị (${data.items.length} chủng loại)!`;
}

// Chỉnh sửa thông tin 1 máy tồn kho (CÓ BẢO VỆ CHỐNG TRÙNG SERIAL)
function updateThietBi(rowId, serial, model, kho, ncc, ghiChu) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const spSheet = ss.getSheetByName("DM_SAN_PHAM");

  const cleanSerial = String(serial).trim().toUpperCase();
  const targetRow = Number(rowId);

  if (!cleanSerial) throw new Error("Mã Serial không được để trống!");

  // 1. QUÉT CHỐNG TRÙNG: Kiểm tra xem Serial này đã có ở dòng khác chưa
  const lastRow = tbSheet.getLastRow();
  if (lastRow > 1) {
    const allSerials = tbSheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (let i = 0; i < allSerials.length; i++) {
      const currentRow = i + 2;
      const existSn = String(allSerials[i][0]).trim().toUpperCase();
      // Nếu trùng mã Serial với một dòng KHÁC dòng đang sửa -> CHẶN NGAY
      if (existSn === cleanSerial && currentRow !== targetRow) {
        throw new Error(`Mã Serial [${cleanSerial}] đã tồn tại trên một thiết bị khác trong hệ thống! Không thể sửa trùng.`);
      }
    }
  }

  // 2. Tìm tên hàng hóa và nhóm hàng tương ứng với model mới
  let tenHang = "", nhomHang = "";
  if (spSheet && spSheet.getLastRow() > 1) {
    const spData = spSheet.getRange(2, 1, spSheet.getLastRow() - 1, 3).getValues();
    for (let r of spData) {
      if (String(r[0]).toUpperCase() === String(model).toUpperCase()) {
        tenHang = r[1];
        nhomHang = r[2];
        break;
      }
    }
  }

  // 3. Ghi đè thông tin an toàn
  tbSheet.getRange(targetRow, 1).setValue(cleanSerial);
  tbSheet.getRange(targetRow, 2, 1, 3).setValues([[model, tenHang, nhomHang]]);
  tbSheet.getRange(targetRow, 6, 1, 2).setValues([[kho, ncc]]);
  tbSheet.getRange(targetRow, 17).setValue(ghiChu);

  return "Cập nhật thiết bị thành công!";
}

}

  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const tonKhoList = [];
  if (tbSheet.getLastRow() > 1) {
    const data = tbSheet.getRange(2, 1, tbSheet.getLastRow() - 1, 10).getValues();
    data.forEach((r, i) => {
      if (String(r[9]).trim() === "Tồn kho") {
        tonKhoList.push({
          rowId: i + 2,
          serial: String(r[0]).trim(),
          model: String(r[1]).trim(),
          tenHang: String(r[2]).trim(),
          nhomHang: String(r[3]).trim(),
          kho: String(r[5]).trim()
        });
      }
    });
  }

  return {
    khachHang: khData.map(r => ({ ten: r[0], sdt: r[1], diaChi: r[2] })),
    goiBaoHanh: goiBhList.length > 0 ? goiBhList : ["3 tháng", "6 tháng", "12 tháng", "24 tháng", "36 tháng"],
    tonKhoList: tonKhoList
  };
}

// 2. Thực thi Xuất kho & Kích hoạt bảo hành (Zero-Error Export)
function executeXuatKho(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  const lsSheet = ss.getSheetByName("LICH_SU_XUAT");

  if (!data.serials || data.serials.length === 0) {
    throw new Error("Chưa có máy nào được chọn để xuất!");
  }

  const cleanSerials = data.serials.map(s => String(s).trim().toUpperCase());
  const allTbData = tbSheet.getRange(2, 1, tbSheet.getLastRow() - 1, 10).getValues();

  // Kiểm tra điều kiện tồn kho thực tế của tất cả serial
  const rowsToUpdate = [];
  cleanSerials.forEach(sn => {
    let found = false;
    for (let i = 0; i < allTbData.length; i++) {
      if (String(allTbData[i][0]).trim().toUpperCase() === sn) {
        if (String(allTbData[i][9]).trim() !== "Tồn kho") {
          throw new Error(`Serial [${sn}] hiện không còn trong kho (Đã xuất hoặc lỗi)!`);
        }
        rowsToUpdate.push(i + 2);
        found = true;
        break;
      }
    }
    if (!found) throw new Error(`Không tìm thấy mã Serial [${sn}] trong hệ thống!`);
  });

  // Tính ngày hết hạn bảo hành: Ngày xuất + Số tháng BH
  const ngayXuatParts = data.ngayXuat.split('-'); // YYYY-MM-DD
  const expDate = new Date(ngayXuatParts[0], ngayXuatParts[1] - 1, ngayXuatParts[2]);
  const soThang = parseInt(data.soThangBh.replace(/\D/g, ''), 10) || 12;
  expDate.setMonth(expDate.getMonth() + soThang);

  const ngayXuatFormat = Utilities.formatDate(new Date(data.ngayXuat), "GMT+7", "dd/MM/yyyy");
  const ngayHetHanFormat = Utilities.formatDate(expDate, "GMT+7", "dd/MM/yyyy");

  // 1. Cập nhật DATA_THIET_BI: Đổi trạng thái sang "Đã xuất" và điền thông tin BH
  rowsToUpdate.forEach(rowId => {
    tbSheet.getRange(rowId, 10).setValue("Đã xuất"); // Cột J: Trạng thái
    tbSheet.getRange(rowId, 11, 1, 6).setValues([[
      ngayXuatFormat,      // Cột K: Ngày xuất
      data.maPhieu,        // Cột L: Mã phiếu xuất
      data.tenKhach,       // Cột M: Tên khách hàng
      data.sdtKhach,       // Cột N: SĐT khách
      soThang,             // Cột O: Số tháng BH
      ngayHetHanFormat     // Cột P: Ngày hết hạn BH
    ]]);
    if (data.ghiChu) tbSheet.getRange(rowId, 17).setValue(data.ghiChu);
  });

  // 2. Ghi nhật ký vào LICH_SU_XUAT
  lsSheet.appendRow([
    data.maPhieu,
    ngayXuatFormat,
    `${data.tenKhach} (${data.sdtKhach})`,
    cleanSerials.length,
    cleanSerials.join(', '),
    data.soThangBh,
    data.ghiChu
  ]);

  return `Xuất kho thành công ${cleanSerials.length} thiết bị! Đã kích hoạt bảo hành.`;
}

// 3. Tra cứu bảo hành toàn diện (Real-time Timeline & Đèn tín hiệu)
function searchBaoHanhData(keyword) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tbSheet = ss.getSheetByName("DATA_THIET_BI");
  if (tbSheet.getLastRow() <= 1) return [];

  const data = tbSheet.getRange(2, 1, tbSheet.getLastRow() - 1, 17).getValues();
  const today = new Date();
  today.setHours(0,0,0,0);

  const results = [];
  data.forEach((r, idx) => {
    // Chỉ quan tâm máy Đã xuất hoặc có hạn BH
    if (r[9] === "Đã xuất" || r[15]) {
      let expDate = null;
      if (r[15] instanceof Date) {
        expDate = r[15];
      } else if (typeof r[15] === 'string' && r[15].includes('/')) {
        const parts = r[15].split('/');
        expDate = new Date(parts[2], parts[1] - 1, parts[0]);
      }

      let diffDays = -999;
      let statusColor = "RED"; // Đỏ: Hết hạn
      let statusText = "Hết hạn";

      if (expDate && !isNaN(expDate.getTime())) {
        expDate.setHours(0,0,0,0);
        diffDays = Math.ceil((expDate - today) / (1000 * 60 * 60 * 24));
        if (diffDays > 30) {
          statusColor = "GREEN"; // Xanh: Còn hạn
          statusText = `Còn ${diffDays} ngày`;
        } else if (diffDays >= 0) {
          statusColor = "YELLOW"; // Vàng: Sắp hết hạn
          statusText = `Còn ${diffDays} ngày (Gấp)`;
        } else {
          statusColor = "RED";
          statusText = `Quá hạn ${Math.abs(diffDays)} ngày`;
        }
      }

      results.push({
        serial: r[0],
        model: r[1],
        tenHang: r[2],
        nhomHang: r[3],
        loaiHang: r[4],
        kho: r[5],
        ncc: r[6],
        ngayNhap: r[7] instanceof Date ? Utilities.formatDate(r[7], "GMT+7", "dd/MM/yyyy") : r[7],
        ngayXuat: r[10] instanceof Date ? Utilities.formatDate(r[10], "GMT+7", "dd/MM/yyyy") : r[10],
        maPhieuXuat: r[11],
        khachHang: r[12],
        sdtKhach: r[13],
        soThangBh: r[14],
        ngayHetHanBh: r[15] instanceof Date ? Utilities.formatDate(r[15], "GMT+7", "dd/MM/yyyy") : r[15],
        diffDays: diffDays,
        statusColor: statusColor,
        statusText: statusText
      });
    }
  });

  return results;
}
